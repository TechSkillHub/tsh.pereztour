# /etc/nginx/sites-available/pereztour.tshdevelop.com
server {
    listen 80;
    server_name pereztour.tshdevelop.com;
}

server {
    listen 443 ssl;
    server_name pereztour.tshdevelop.com;

    # Configuração do Nuxt.js (Static)
    root /var/www/pereztour.tshdevelop.com/dist;
    index index.html;

    # Configuração do Nuxt.js (SSR - descomente se necessário)
    # location / {
    #     proxy_pass http://localhost:3000;
    #     proxy_http_version 1.1;
    #     proxy_set_header Upgrade $http_upgrade;
    #     proxy_set_header Connection 'upgrade';
    #     proxy_set_header Host $host;
    #     proxy_cache_bypass $http_upgrade;
    # }

    # Configuração para Static Site (descomente se for estático)
    location / {
        try_files $uri $uri/ /index.html;
        # IMPORTANTE para carregar sections/assets
        add_header 'Cache-Control' 'no-store, no-cache, must-revalidate, proxy-revalidate, max-age=0';
        if_modified_since off;
        expires off;
        etag off;
    }

    # Otimizações
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;

    # Cache para assets
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg)$ {
        expires 1y;
        add_header Cache-Control "public";
        access_log off;
    }

    # Error pages
    error_page 404 /404.html;
    error_page 500 502 503 504 /500.html;

    # Bloqueia acesso a arquivos ocultos
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }

    # Logs
    access_log /var/log/nginx/pereztour.tshdevelop.com.access.log;
    error_log /var/log/nginx/pereztour.tshdevelop.com.error.log;
}
